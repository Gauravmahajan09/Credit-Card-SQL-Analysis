-- ==========================================
-- 5. BASIC ANALYSIS
-- ==========================================

-- Total transaction amount
SELECT SUM(amount) AS total_transaction_amount
FROM transactions;

-- Average transaction amount
SELECT AVG(amount) AS average_transaction_amount
FROM transactions;

-- Highest transaction
SELECT MAX(amount) AS highest_transaction
FROM transactions;

-- Lowest transaction
SELECT MIN(amount) AS lowest_transaction
FROM transactions;


-- ==========================================
-- 6. SPENDING BY CATEGORY
-- ==========================================

SELECT
    m.category,
    SUM(t.amount) AS total_spending
FROM transactions t
JOIN merchants m
    ON t.merchant_id = m.merchant_id
WHERE t.payment_status = 'Success'
GROUP BY m.category
ORDER BY total_spending DESC;


-- ==========================================
-- 7. CUSTOMER SPENDING ANALYSIS
-- ==========================================

SELECT
    c.customer_id,
    c.customer_name,
    SUM(t.amount) AS total_spending
FROM customers c
JOIN credit_cards cc
    ON c.customer_id = cc.customer_id
JOIN transactions t
    ON cc.card_id = t.card_id
WHERE t.payment_status = 'Success'
GROUP BY c.customer_id, c.customer_name
ORDER BY total_spending DESC;


-- ==========================================
-- 8. TOP 3 CUSTOMERS
-- ==========================================

SELECT
    c.customer_name,
    SUM(t.amount) AS total_spending
FROM customers c
JOIN credit_cards cc
    ON c.customer_id = cc.customer_id
JOIN transactions t
    ON cc.card_id = t.card_id
WHERE t.payment_status = 'Success'
GROUP BY c.customer_id, c.customer_name
ORDER BY total_spending DESC
LIMIT 3;


-- ==========================================
-- 9. PAYMENT STATUS ANALYSIS
-- ==========================================

SELECT
    payment_status,
    COUNT(*) AS transaction_count
FROM transactions
GROUP BY payment_status;


-- ==========================================
-- 10. SPENDING BY CARD TYPE
-- ==========================================

SELECT
    cc.card_type,
    SUM(t.amount) AS total_spending
FROM credit_cards cc
JOIN transactions t
    ON cc.card_id = t.card_id
WHERE t.payment_status = 'Success'
GROUP BY cc.card_type
ORDER BY total_spending DESC;


-- ==========================================
-- 11. CUSTOMER SEGMENTATION
-- ==========================================

SELECT
    c.customer_name,
    SUM(t.amount) AS total_spending,
    CASE
        WHEN SUM(t.amount) >= 15000 THEN 'High Spender'
        WHEN SUM(t.amount) >= 5000 THEN 'Medium Spender'
        ELSE 'Low Spender'
    END AS spending_category
FROM customers c
JOIN credit_cards cc
    ON c.customer_id = cc.customer_id
JOIN transactions t
    ON cc.card_id = t.card_id
WHERE t.payment_status = 'Success'
GROUP BY c.customer_id, c.customer_name;


-- ==========================================
-- 12. CUSTOMER SPENDING RANK
-- ==========================================

SELECT
    c.customer_name,
    SUM(t.amount) AS total_spending,
    RANK() OVER (
        ORDER BY SUM(t.amount) DESC
    ) AS spending_rank
FROM customers c
JOIN credit_cards cc
    ON c.customer_id = cc.customer_id
JOIN transactions t
    ON cc.card_id = t.card_id
WHERE t.payment_status = 'Success'
GROUP BY c.customer_id, c.customer_name;


-- ==========================================
-- 13. HIGHEST TRANSACTION BY CATEGORY
-- ==========================================

WITH category_transactions AS (
    SELECT
        m.category,
        t.transaction_id,
        t.amount,
        ROW_NUMBER() OVER (
            PARTITION BY m.category
            ORDER BY t.amount DESC
        ) AS rn
    FROM transactions t
    JOIN merchants m
        ON t.merchant_id = m.merchant_id
    WHERE t.payment_status = 'Success'
)
SELECT *
FROM category_transactions
WHERE rn = 1;


-- ==========================================
-- 14. CUSTOMERS ABOVE AVERAGE SPENDING
-- ==========================================

WITH customer_spending AS (
    SELECT
        c.customer_id,
        c.customer_name,
        SUM(t.amount) AS total_spending
    FROM customers c
    JOIN credit_cards cc
        ON c.customer_id = cc.customer_id
    JOIN transactions t
        ON cc.card_id = t.card_id
    WHERE t.payment_status = 'Success'
    GROUP BY c.customer_id, c.customer_name
)
SELECT *
FROM customer_spending
WHERE total_spending > (
    SELECT AVG(total_spending)
    FROM customer_spending
);


-- ==========================================
-- 15. HIGH VALUE TRANSACTIONS
-- ==========================================

SELECT
    t.transaction_id,
    c.customer_name,
    t.amount,
    m.merchant_name,
    m.category
FROM transactions t
JOIN credit_cards cc
    ON t.card_id = cc.card_id
JOIN customers c
    ON cc.customer_id = c.customer_id
JOIN merchants m
    ON t.merchant_id = m.merchant_id
WHERE t.amount > (
    SELECT AVG(amount) * 2
    FROM transactions
)
AND t.payment_status = 'Success';
